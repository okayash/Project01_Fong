Optimimzed Query for ec

SELECT P.Pnumber, P.Pname, COUNT(E.Ssn), SUM(W.Hours)
FROM PROJECT P
INNER JOIN WORKSON W ON P.Pnumber = W.Pno
INNER JOIN EMPLOYEE E ON E.Ssn = W.Essn
WHERE P.Plocation = 'Dallas' AND (E.Sex = 'F') AND (E.Salary >= 70000)
GROUP BY P.Pnumber, P.Pname
HAVING SUM(W.Hours) > 100
ORDER BY P.Pname DESC;
